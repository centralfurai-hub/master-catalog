Master Catalog × Supabase
Hubungkan ke database Supabase Anda
Cara setup: Buka supabase.com → buat project → copy Project URL dan anon key → paste di form ini dan klik Hubungkan
📋 SQL — jalankan sekali di Supabase SQL Editor
Buka Supabase → SQL Editor → paste semua SQL di bawah → klik Run. Sudah termasuk setup tabel + bucket foto + keamanan RLS.
⚠️ Setelah ini, buat akun login resmi di Authentication → Users (email owner@internal.mastercatalog & spg@internal.mastercatalog), lalu copy UID akun Owner dan ganti GANTI_DENGAN_UUID_OWNER di bawah, lalu jalankan ulang bagian POLICY-nya.
-- TABEL CREATE TABLE IF NOT EXISTS mc_products ( id TEXT PRIMARY KEY, name TEXT NOT NULL, alias TEXT, harga BIGINT DEFAULT 0, kategori TEXT, keterangan TEXT, divisi JSONB DEFAULT '{"Global":0,"Central":0,"Celine":0,"Panda":0,"Titipan":0,"Servicean":0}', img TEXT, created_at TEXT, updated_at TEXT ); -- INDEX untuk pencarian cepat berdasarkan nama CREATE INDEX IF NOT EXISTS idx_mc_products_name ON mc_products(name); CREATE TABLE IF NOT EXISTS mc_categories (id SERIAL PRIMARY KEY, name TEXT UNIQUE NOT NULL); CREATE TABLE IF NOT EXISTS mc_log (id SERIAL PRIMARY KEY, msg TEXT, type TEXT, username TEXT, created_at TEXT); -- KEAMANAN: AKTIFKAN RLS (jangan dimatikan!) ALTER TABLE mc_products ENABLE ROW LEVEL SECURITY; ALTER TABLE mc_categories ENABLE ROW LEVEL SECURITY; ALTER TABLE mc_log ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "mc_products_select" ON mc_products; DROP POLICY IF EXISTS "mc_products_write" ON mc_products; DROP POLICY IF EXISTS "mc_categories_select" ON mc_categories; DROP POLICY IF EXISTS "mc_categories_write" ON mc_categories; DROP POLICY IF EXISTS "mc_log_select" ON mc_log; DROP POLICY IF EXISTS "mc_log_insert" ON mc_log; -- Baca katalog: terbuka untuk semua (termasuk Guest/tamu tanpa login) CREATE POLICY "mc_products_select" ON mc_products FOR SELECT TO anon, authenticated USING (true); CREATE POLICY "mc_categories_select" ON mc_categories FOR SELECT TO anon, authenticated USING (true); -- Tulis (tambah/ubah/hapus): HANYA akun Owner yang login resmi -- Ganti GANTI_DENGAN_UUID_OWNER dengan UID asli dari Authentication > Users CREATE POLICY "mc_products_write" ON mc_products FOR ALL TO authenticated USING (auth.uid() = 'GANTI_DENGAN_UUID_OWNER'::uuid) WITH CHECK (auth.uid() = 'GANTI_DENGAN_UUID_OWNER'::uuid); CREATE POLICY "mc_categories_write" ON mc_categories FOR ALL TO authenticated USING (auth.uid() = 'GANTI_DENGAN_UUID_OWNER'::uuid) WITH CHECK (auth.uid() = 'GANTI_DENGAN_UUID_OWNER'::uuid); -- Log aktivitas: hanya yang login yang boleh baca & tulis CREATE POLICY "mc_log_select" ON mc_log FOR SELECT TO authenticated USING (true); CREATE POLICY "mc_log_insert" ON mc_log FOR INSERT TO authenticated WITH CHECK (true); -- STORAGE BUCKET FOTO INSERT INTO storage.buckets (id, name, public) VALUES ('mc-photos', 'mc-photos', true) ON CONFLICT (id) DO UPDATE SET public = true; -- POLICY: lihat foto — boleh publik (foto produk memang ditampilkan ke semua) DROP POLICY IF EXISTS "mc_photos_select" ON storage.objects; CREATE POLICY "mc_photos_select" ON storage.objects FOR SELECT TO anon, authenticated USING (bucket_id = 'mc-photos'); -- POLICY: upload foto baru — HANYA yang login (Owner), bukan publik/anon lagi DROP POLICY IF EXISTS "mc_photos_insert" ON storage.objects; CREATE POLICY "mc_photos_insert" ON storage.objects FOR INSERT TO authenticated WITH CHECK (bucket_id = 'mc-photos'); -- POLICY: ganti/replace foto — HANYA yang login (Owner), bukan publik/anon lagi DROP POLICY IF EXISTS "mc_photos_update" ON storage.objects; CREATE POLICY "mc_photos_update" ON storage.objects FOR UPDATE TO authenticated USING (bucket_id = 'mc-photos');
